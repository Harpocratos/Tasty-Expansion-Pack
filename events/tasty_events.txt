				add_deposit = d_ore_rich_caverns
				add_deposit = d_ore_rich_caverns
				add_deposit = d_submerged_ore_veins
                # Clear pre-existing pops from any prior game start step

                    kill_pop_group = { pop_group = THIS percentage = 0.8 }

                remove_all_districts = yes

					count = 2

        }

        every_country = {

            }
            capital_scope = {

				}

            }
            capital_scope = {

                    kill_pop_group = { pop_group = THIS percentage = 0.8 }
