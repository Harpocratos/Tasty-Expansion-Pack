zone_nucleosynthesis = {
	icon = GFX_colony_type_foundry # Placeholder icon
	base_buildtime = 400

    potential = { # planet scope
        exists = owner
        owner = { has_civic = civic_violet_flood }
        OR = {
            is_planet_class = pc_nanomass
            is_planet_class = pc_nanoplasma
        }
    }

	unlock = { # planet scope
		always = yes
	}

	resources = {
		category = planet_districts # Should be planet_zones or similar if costing
		cost = {
			minerals = 100 # Example cost to build the zone itself
            alloys = 500
		}
	}

	# Buildings that can be constructed in the site provided by the district's zone slot
	include = {
		building_atomic_nucleosynthesis_pool
		building_bioreticulator
	}

	planet_modifier = {
		zone_building_slots_add = 0 # This zone type does not add new planet-wide building slots
	}

    # This triggered modifier on the zone itself is unusual, usually it's on the district or buildings
    triggered_district_planet_modifier = {
        potential = {
            exists = owner
            owner = { has_civic = civic_violet_flood }
            OR = {
                is_planet_class = pc_nanomass
                is_planet_class = pc_nanoplasma
            }
        }
        modifier = { # These are effectively direct bonuses from building the zone
            job_nucleosynthesis_pool_add = 100 # 1 job
            planet_housing_add = 100 # 1 housing
        }
    }
}
